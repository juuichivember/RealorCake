import pygame
import os

# กำหนดค่าพื้นฐาน
pygame.init()
WIDTH, HEIGHT = 1920, 1080  # ขนาดหน้าจอแบบ Full HD
screen = pygame.display.set_mode((WIDTH, HEIGHT), pygame.FULLSCREEN)
pygame.display.set_caption("RealOrCake - Select Your Cake Base")

# โหลดภาพพื้นหลัง
background = pygame.image.load("Elements/shop_background.png")
background = pygame.transform.scale(background, (WIDTH, HEIGHT))

# โหลดภาพชั้นวางเค้ก
shelve = pygame.image.load("Elements/shelve.png")
shelve = pygame.transform.scale(shelve, (775, 806))  # ปรับขนาดชั้นวาง
shelve_x, shelve_y = (1083, 138)  # แสดงชั้นวางเลย ไม่ต้องเคลื่อน

# โหลดปุ่มรีเซ็ตและปุ่มเสร็จสิ้น
reset_button = pygame.image.load("Elements/reset_button.png")
reset_button = pygame.transform.scale(reset_button, (212, 124))
reset_button_pos = (1240, 940)
reset_rect = reset_button.get_rect(topleft=reset_button_pos)

show_button = pygame.image.load("Elements/show_button.png")
show_button = pygame.transform.scale(show_button, (212, 124))
show_button_pos = (1540, 940)
show_rect = show_button.get_rect(topleft=show_button_pos)

# โหลดปุ่มกลับหน้า Home
back_button = pygame.image.load("Elements/back_button.png")
back_button = pygame.transform.scale(back_button, (222, 122))
back_button_pos = (50, 50)
back_rect = back_button.get_rect(topleft=back_button_pos)

# โหลดถาดสี (Palette)
palette_bg = pygame.image.load("Elements/Color/painttray.png")
palette_bg = pygame.transform.scale(palette_bg, (150, 390))
palette_pos = (930, 150)

# รายชื่อสีในพาเลตต์
color_names = ["none", "grape", "bluberry", "mint", "vanilla", "milk", "carrot", "redvelvet", "strawberry", "charcole", "chocolate", "coffee"]

# โหลดไอคอนสี
color_icons = {color: pygame.image.load(f"Elements/Color/{color}.png") for color in color_names}

# ปรับขนาดไอคอนสี
for key in color_icons:
    color_icons[key] = pygame.transform.scale(color_icons[key], (50, 50))

# ตำแหน่งของไอคอนสี (2 คอลัมน์)
color_positions = []
for i in range(len(color_names)):
    col = i % 2  # คำนวณคอลัมน์ (0 หรือ 1)
    row = i // 2  # คำนวณแถว
    x_pos = palette_pos[0] + 20 + (col * 60)  # คอลัมน์ห่างกัน 60 px
    y_pos = palette_pos[1] + 20 + (row * 60)  # แถวห่างกัน 60 px
    color_positions.append((x_pos, y_pos))

# เก็บสีที่เลือก
selected_color = None

# กำหนดสี
WHITE = (255, 255, 255)
BLACK = (0, 0, 0)

# ฟังก์ชันโหลดรูปภาพจากโฟลเดอร์
def load_cake_image(cake_type):
    path = f"Elements/layer7_base_1/layer7_base_1{cake_type}.png"
    if os.path.exists(path):
        return pygame.image.load(path)
    return None

# รายการชนิดเค้ก
cake_types = [
    "vanilla", "chocolate", "strawberry",
    "charcole", "blueberry", "grape",
    "carrot", "milk", "mint"
]

# โหลดภาพเค้กจากโฟลเดอร์
cake_options = {cake: load_cake_image(cake) for cake in cake_types}

# ตำแหน่งของเค้ก
final_positions = {
    "vanilla": (1122, 160), "chocolate": (1339, 160), "strawberry": (1556, 160),
    "charcole": (1122, 394), "blueberry": (1339, 394), "grape": (1556, 394),
    "carrot": (1122, 636), "milk": (1339, 636), "mint": (1556, 636)
}

for key in cake_options:
    if cake_options[key]:  # ตรวจสอบว่ารูปโหลดสำเร็จ
        cake_options[key] = pygame.transform.scale(cake_options[key], (260, 260))

# ฐานเค้กที่ถูกเลือก
selected_cake = None

# ฟอนต์
font = pygame.font.Font(None, 50)

# ตัวแปรควบคุมการแสดงเค้ก
show_cakes = True  # ปรับให้เค้กแสดงผลทันที

# วนลูปเกม
running = True
while running:
    screen.blit(background, (0, 0))  # วาดพื้นหลัง
    screen.blit(palette_bg, palette_pos)  # วาดพาเลตต์สี
    screen.blit(shelve, (shelve_x, shelve_y))
    screen.blit(back_button, back_button_pos)

    # วาดไอคอนสี
    for i, color in enumerate(color_names):
        screen.blit(color_icons[color], color_positions[i])

    screen.blit(shelve, (shelve_x, shelve_y))
    screen.blit(back_button, back_button_pos)

    # แสดงข้อความ
    text = font.render("Select Your Cake Base", True, BLACK)
    screen.blit(text, (WIDTH // 2 - text.get_width() // 2, 100))

    # แสดงฐานเค้ก
    if show_cakes:
        for key, pos in final_positions.items():
            if cake_options[key]:
                screen.blit(cake_options[key], pos)
        screen.blit(reset_button, reset_button_pos)
        screen.blit(show_button, show_button_pos)

    # แสดงฐานเค้กที่ถูกเลือก
    if selected_cake:
        selected_cake_img = load_cake_image(selected_cake)
        if selected_cake_img:
            selected_cake_img = pygame.transform.scale(selected_cake_img, (847, 847))
            cake_x = 100
            cake_y = 248
            screen.blit(selected_cake_img, (cake_x, cake_y))

    # ตรวจจับเหตุการณ์
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False

        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_ESCAPE:  # กด ESC เพื่อออกจากเกม
                running = False

        if event.type == pygame.MOUSEBUTTONDOWN and show_cakes:
            x, y = event.pos
            # ตรวจจับการกดเลือกสีจากพาเลตต์
            for i, color_pos in enumerate(color_positions):
                color_rect = pygame.Rect(color_pos[0], color_pos[1], 50, 50)
                if color_rect.collidepoint(x, y):
                    selected_color = color_names[i]  # บันทึกสีที่เลือก
                    print(f"Selected color: {selected_color}")

    pygame.display.flip()  # อัปเดตหน้าจอ

pygame.quit()
